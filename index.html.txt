/* =======================

   ======================= */
const FACTS = [
  "There are more possible iterations of a game of chess than atoms in the observable universe (estimate: ~10^120).",
  "Every element heavier than helium was forged inside stars or during explosive stellar events.",
  "Our brain's neurons transmit signals via both electrical impulses and chemical neurotransmitters.",
  "CRISPR is a gene-editing technique that can precisely alter DNA sequences in living cells.",
  "The observable universe is about 93 billion light-years across.",
  "Water is the only common substance on Earth that expands when it freezes, which is crucial for aquatic life.",
  "Time passes more slowly near massive objects — this is gravitational time dilation from General Relativity.",
  "Plants can communicate via chemical signals through the air and by electrical/chemical messages in root networks.",
  "The human genome has roughly 3.1 billion base pairs and yet we share about 99.9% of it with other humans.",
  "Photosynthesis converts sunlight into chemical energy with an efficiency that biology has optimized over billions of years."
];

const INTERPRETATIONS = [
  "Growth and transformation often take place where energy meets structure — like stars forging new elements.",
  "Even the smallest change in a system (like a gene edit) can ripple outward; intention matters.",
  "Knowing that time is flexible under gravity invites humility about our place and perspective.",
  "The web of life is communicative and cooperative, suggesting interconnectedness beneath apparent separateness.",
  "Complexity can emerge from simple rules — small daily choices add up to surprising transformation.",
  "Curiosity is an engine of meaning; the universe offers patterns waiting for attention.",
  "Silence and stillness can be like fertile dark space where new insights are born.",
  "Our inner wiring is both biological and story-based; reshaping habits changes both brain and narrative.",
  "The cosmos is both indifferent and generous — its laws don’t care, but they allow life to flourish.",
  "Observing with wonder changes the observer — noticing creates a kind of creative participation."
];

const QUESTIONS = [
  "What small action today might change the shape of your future story?",
  "Where do you feel most connected to something larger than yourself?",
  "What belief of yours could benefit from a gentle revision based on new evidence?",
  "If you had to explain wonder to someone who'd never seen the night sky, what would you say?",
  "What pattern in your life repeats, and what might it be trying to teach you?",
  "When have you experienced a tiny change that led to a big result?",
  "What would you pay attention to today if you knew your perception shapes reality?",
  "How do scientific facts change the way you interpret a spiritual feeling?",
  "What part of you resists change, and what part quietly invites it?",
  "Where can curiosity replace certainty in your next decision?"
];

/* =======================
   App logic
   ======================= */

const el = id => document.getElementById(id);
const factEl = el('fact');
const interpretEl = el('interpret');
const questionEl = el('question');
const revealBtn = el('reveal');
const anotherBtn = el('another');
const saveBtn = el('save');
const copyBtn = el('copy');
const favoritesList = el('favoritesList');
const recentMeta = el('recentMeta');
const clearFavBtn = el('clearFav');

let current = null;
let favorites = JSON.parse(localStorage.getItem('bridge_favs') || "[]");

/* utility: random pick */
function pick(arr){ return arr[Math.floor(Math.random()*arr.length)]; }

/* format and show */
function buildInsight(){
  const fact = pick(FACTS);
  const interpretation = pick(INTERPRETATIONS);
  const question = pick(QUESTIONS);
  return { fact, interpretation, question, ts: new Date().toISOString() };
}

function showInsight(obj, animate=true){
  current = obj;
  factEl.textContent = obj.fact;
  interpretEl.textContent = obj.interpretation;
  questionEl.textContent = obj.question;

  // reveal pieces with slight stagger
  const pieces = [factEl, interpretEl, questionEl];
  pieces.forEach(p => p.classList.remove('show'));
  if(!animate){
    pieces.forEach(p => p.classList.add('show'));
  } else {
    pieces.forEach((p,i) => {
      setTimeout(()=>p.classList.add('show'), 120 + i*120);
    });
  }

  recentMeta.textContent = `Fact · ${new Date(obj.ts).toLocaleString()}`;
}

function revealAction(){
  const ins = buildInsight();
  showInsight(ins, true);
  // small UI flourish
  revealBtn.classList.remove('pulse');
  setTimeout(()=>revealBtn.classList.add('pulse'), 900);
}

/* Save to favorites */
function saveCurrent(){
  if(!current) return;
  favorites.unshift(current);
  // keep a reasonable cap
  if(favorites.length > 80) favorites = favorites.slice(0,80);
  localStorage.setItem('bridge_favs', JSON.stringify(favorites));
  renderFavorites();
}

/* render favorites list */
function renderFavorites(){
  favoritesList.innerHTML = '';
  if(favorites.length === 0){
    favoritesList.innerHTML = '<div class="small" style="opacity:.7">No favorites yet. Save your favorite insights with the Save button.</div>';
    return;
  }
  favorites.forEach((f, idx) => {
    const div = document.createElement('div');
    div.className = 'favItem';
    div.innerHTML = `<div style="font-weight:700">${f.fact}</div>
                     <div style="font-size:13px;color:var(--muted);margin-top:6px">${f.interpretation}</div>
                     <div style="font-size:12px;margin-top:8px;color:var(--muted)"><em>${f.question}</em></div>
                     <div style="margin-top:8px;display:flex;gap:8px;align-items:center">
                       <button class="btn" data-index="${idx}" data-action="use">Use</button>
                       <button class="btn" data-index="${idx}" data-action="remove">Remove</button>
                       <div style="flex:1"></div>
                       <div class="small">${new Date(f.ts).toLocaleDateString()}</div>
                     </div>`;
    favoritesList.appendChild(div);
  });
}

/* handle favorites click */
favoritesList.addEventListener('click', (ev)=>{
  const btn = ev.target.closest('button');
  if(!btn) return;
  const idx = Number(btn.dataset.index);
  const action = btn.dataset.action;
  if(action === 'use'){
    const item = favorites[idx];
    if(item) showInsight(item, false);
  } else if(action === 'remove'){
    favorites.splice(idx,1);
    localStorage.setItem('bridge_favs', JSON.stringify(favorites));
    renderFavorites();
  }
});

/* copy to clipboard */
async function copyCurrent(){
  if(!current) return;
  const text = `${current.fact}\n\nInterpretation: ${current.interpretation}\n\nReflection: ${current.question}`;
  try {
    await navigator.clipboard.writeText(text);
    copyBtn.textContent = 'Copied ✓';
    setTimeout(()=>copyBtn.textContent = 'Copy',1200);
  } catch(e){
    // fallback
    const ta = document.createElement('textarea');
    ta.value = text; document.body.appendChild(ta);
    ta.select();
    document.execCommand('copy');
    ta.remove();
    copyBtn.textContent = 'Copied ✓';
    setTimeout(()=>copyBtn.textContent = 'Copy',1200);
  }
}

/* clear favorites */
clearFavBtn.addEventListener('click', ()=>{
  if(!confirm('Clear all saved favorites?')) return;
  favorites = [];
  localStorage.removeItem('bridge_favs');
  renderFavorites();
});

/* hook up buttons */
revealBtn.addEventListener('click', revealAction);
